    1 & 
    Chỉnh sửa thông tin tổ chức & 
    Tiền điều kiện:
    \begin{itemize}
        \item Đã đăng nhập vào hệ thống với tư cách Chủ sở hữu
        \item Đã nhấn chọn tab "Tổ chức" trên màn hình
    \end{itemize}
    Các bước thực hiện: 
    \begin{enumerate}[label=\arabic*.]
        \item Chọn Chỉnh sửa
        \item Thay đổi các trường và chọn Lưu
    \end{enumerate}
    & 
    \begin{enumerate}[label=\arabic*.]
        \item Màn hình cho phép tùy chỉnh các trường được phép chỉnh sửa
        \item Màn hình hiển thị thông báo thành công và các thông tin được cập nhật
    \end{enumerate}  & Đạt \\

        \item Đã nhấn chọn tab "Tổ chức" trên màn hình

    3 & 
    Xóa thành viên & 
    Tiền điều kiện:
    \begin{itemize}
        \item Đã đăng nhập vào hệ thống với tư cách Chủ sở hữu
        \item Đang ở mục Thành viên trong tab "Tổ chức"
        \item Danh sách thành viên tổ chức có ít nhất một thành viên không phải chủ sở hữu
    \end{itemize}
    Các bước thực hiện: 
    \begin{enumerate}[label=\arabic*.]
        \item Chọn Xóa một thành viên không phải Chủ sở hữu
        \item Chọn Xác nhận
    \end{enumerate}
    & 
    \begin{enumerate}[label=\arabic*.]
        \item Màn hình hiển thị cửa sổ xác nhận xóa
        \item Màn hình hiển thị thông báo thành công và thành viên bị xóa khỏi tổ chức, đồng thời tài khoản của thành viên này sẽ bị vô hiệu hóa.
    \end{enumerate}  & Đạt \\

        \item Đang ở mục Thành viên trong tab "Tổ chức"

    \begin{enumerate}[label=\arabic*.]

    \end{enumerate}  & Đạt \\

    5 & 
    Hủy lời mời đã gửi & 
    Tiền điều kiện:
    \begin{itemize}
        \item Đã đăng nhập vào hệ thống với tư cách Chủ sở hữu
        \item Đang ở mục Lời mời đã gửi trong tab "Tổ chức"
        \item Danh sách lời mời có ít nhất một lời mời ở trạng thái Đang chờ
    \end{itemize}
    Các bước thực hiện: 
    \begin{enumerate}[label=\arabic*.]
        \item Chọn Hủy một lời mời có trạng thái Đang chờ
        \item Chọn Xác nhận
    \end{enumerate}
    & 
    \begin{enumerate}[label=\arabic*.]
        \item Màn hình hiển thị cửa sổ xác nhận
        \item Màn hình hiển thị thông báo thành công và lời mời chuyển sang trạng thái Đã hủy, người được mời sẽ không thể dùng thông tin đã được gửi qua email trước đó để đăng ký tài khoản.
    \end{enumerate}  & Đạt \\

    Mời thêm thành viên &

        \item Đang ở mục Mời thêm thành viên trong tab "Tổ chức"
    \end{itemize}
    Các bước thực hiện: 
    \begin{enumerate}[label=\arabic*.]
        \item Chưa điền gì chọn Mời
        \item Điền email, vai trò của người được mời và chọn Lưu
    \end{enumerate}
    & 
    \begin{enumerate}[label=\arabic*.]
        \item Màn hình hiển thị lỗi khi chưa điền thông tin
        \item Màn hình hiển thị thông báo thành công và người nhận sẽ nhận được một thư về thông tin đăng ký qua địa chỉ email đã nhập
    \end{enumerate}  & Đạt \\
