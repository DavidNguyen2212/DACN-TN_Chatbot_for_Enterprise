\textbf{Các lớp cơ sở (Base Classes)}

\begin{itemize}
    \item \textbf{BaseEntity}
    \begin{itemize}
        \item \textit{Mục đích}: Là lớp trừu tượng cơ sở cho tất cả các entity, đảm bảo tính nhất quán của dữ liệu
        \item \textit{Thuộc tính chính}:
        \begin{itemize}
            \item id (UUID): Định danh cho mỗi entity
            \item createdAt (DateTime): Thời điểm tạo bản ghi
            \item updatedAt (DateTime): Thời điểm cập nhật cuối cùng
            \item isActive (boolean): Trạng thái hoạt động của entity
        \end{itemize}
        \item \textit{Chức năng chính}:
        \begin{itemize}
            \item Theo dõi lịch sử thay đổi cơ bản
            \item Cung cấp phương thức xác thực cơ bản
        \end{itemize}
    \end{itemize}

    \item \textbf{BaseUserEntity}
    \begin{itemize}
        \item \textit{Mục đích}: Quản lý thông tin cơ bản của người dùng trong hệ thống
        \item \textit{Thuộc tính chính}:
        \begin{itemize}
            \item email (String): Địa chỉ email đăng nhập
            \item passwordHash (String): Mật khẩu đã được mã hóa
            \item fullName (String): Họ tên đầy đủ
            \item lastLoginAt (DateTime): Thời điểm đăng nhập gần nhất
            \item failedLoginAttempts (int): Số lần đăng nhập thất bại
            \item status (UserStatus): Trạng thái tài khoản
        \end{itemize}
        \item \textit{Chức năng chính}:
        \begin{itemize}
            \item Xác thực email và mật khẩu
            \item Quản lý trạng thái đăng nhập
            \item Cung cấp các phương thức bảo mật
        \end{itemize}
    \end{itemize}
\end{itemize}

\textbf{Các lớp quản lý công ty và người dùng}

\begin{itemize}
    \item \textbf{CompanyEntity}
    \begin{itemize}
        \item \textit{Mục đích}: Quản lý thông tin và cấu hình của các công ty sử dụng hệ thống
        \item \textit{Thuộc tính chính}:
        \begin{itemize}
            \item name (String): Tên công ty
            \item domain (String): Tên miền công ty
            \item logo (Image): Logo công ty
            \item subscriptionType (SubscriptionType): Loại gói đăng ký
            \item verificationStatus (VerificationStatus): Trạng thái xác thực
            \item contactInfo (ContactInformation): Thông tin liên hệ
            \item settings (CompanySettings): Cấu hình riêng
            \item billingInfo (BillingInformation): Thông tin thanh toán
            \item maxChatbots (Integer): Số lượng chatbot tối đa
        \end{itemize}
        \item \textit{Quan hệ (Relationships)}:
        \begin{itemize}
            \item chatbots (Set<ChatbotEntity>): Danh sách chatbot
            \item agents (Set<AgentEntity>): Danh sách agent
            \item admins (Set<SystemAdminEntity>): Danh sách admin
        \end{itemize}
        \item \textit{Chức năng chính}:
        \begin{itemize}
            \item Quản lý cấu hình và giới hạn sử dụng
            \item Kiểm soát quyền truy cập và xác thực
            \item Quản lý thanh toán và gói đăng ký
        \end{itemize}
    \end{itemize}

    \item \textbf{SystemAdminEntity}
    \begin{itemize}
        \item \textit{Mục đích}: Quản lý các quản trị viên của hệ thống
        \item \textit{Thuộc tính chính}:
        \begin{itemize}
            \item companyId (UUID): ID công ty quản lý
            \item permissions (Set<Permission>): Các quyền hạn trong hệ thống
            \item lastActivityAt (DateTime): Thời điểm hoạt động cuối
            \item loginHistory (List<LoginRecord>): Lịch sử đăng nhập
        \end{itemize}
        \item \textit{Chức năng chính}:
        \begin{itemize}
            \item Quản lý quyền hạn và phân quyền
            \item Theo dõi hoạt động trong hệ thống
        \end{itemize}
    \end{itemize}

    \item \textbf{AgentEntity}
    \begin{itemize}
        \item \textit{Mục đích}: Quản lý thông tin và hoạt động của nhân viên chăm sóc khách hàng
        \item \textit{Thuộc tính chính}:
        \begin{itemize}
            \item companyId (UUID): ID công ty
            \item status (AgentStatus): Trạng thái làm việc
            \item maxConcurrentChats (int): Số cuộc hội thoại tối đa
            \item activeChats (int): Số cuộc hội thoại hiện tại
            \item availability (AvailabilityStatus): Trạng thái khả dụng
        \end{itemize}
        \item \textit{Quan hệ}:
        \begin{itemize}
            \item schedule (ScheduleEntity): Lịch làm việc
            \item conversations (Set<ConversationEntity>): Cuộc hội thoại
        \end{itemize}
        \item \textit{Chức năng chính}:
        \begin{itemize}
            \item Quản lý trạng thái và khả dụng
            \item Điều phối các cuộc hội thoại
            \item Quản lý lịch làm việc và nghỉ phép
        \end{itemize}
    \end{itemize}
\end{itemize}

\textbf{Các lớp quản lý lịch làm việc}

\begin{itemize}
    \item \textbf{ScheduleEntity}
    \begin{itemize}
        \item \textit{Mục đích}: Quản lý lịch làm việc của nhân viên chăm sóc khách hàng
        \item \textit{Thuộc tính chính}:
        \begin{itemize}
            \item agentId (UUID): ID của nhân viên
            \item timezone (String): Múi giờ làm việc
            \item defaultHours (WorkingHours): Giờ làm việc mặc định
        \end{itemize}
        \item \textit{Chức năng chính}:
        \begin{itemize}
            \item Quản lý ca làm việc, hỗ trợ sắp xếp đặt lịch hẹn với nhân viên 
            \item Xử lý ngày nghỉ phép
        \end{itemize}
    \end{itemize}
\end{itemize}

\textbf{Các lớp quản lý Chatbot}

\begin{itemize}
    \item \textbf{ChatbotEntity}
    \begin{itemize}
        \item \textit{Mục đích}: Quản lý thông tin và cấu hình của các chatbot trong hệ thống
        \item \textit{Thuộc tính chính}:
        \begin{itemize}
            \item companyId (UUID): ID công ty sở hữu
            \item name (String): Tên chatbot
            \item description (String): Mô tả chức năng
            \item type (ChatbotType): Loại chatbot (FAQ, Customer Service, etc.)
            \item status (ChatbotStatus): Trạng thái hoạt động
            \item trainingStatus (TrainingStatus): Trạng thái huấn luyện
            \item personality (PersonalityConfig): Cấu hình tính cách
            \item appearance (AppearanceConfig): Cấu hình giao diện
            \item settings (ChatbotSettings): Thiết lập chung
            \item integrationConfig (IntegrationConfig): Cấu hình tích hợp
            \item metrics (ChatbotMetrics): Các chỉ số hoạt động
        \end{itemize}
        \item \textit{Quan hệ}:
        \begin{itemize}
            \item knowledgeBases (Set<KnowledgeBaseEntity>): Cơ sở kiến thức
            \item conversations (Set<ConversationEntity>): Lịch sử hội thoại
        \end{itemize}
        \item \textit{Chức năng chính}:
        \begin{itemize}
            \item Quản lý trạng thái và cấu hình chatbot
            \item Theo dõi hiệu suất chatbot
            \item Theo dõi quá trình huấn luyện
        \end{itemize}
    \end{itemize}

    \item \textbf{KnowledgeBaseEntity}
    \begin{itemize}
        \item \textit{Mục đích}: Quản lý nguồn kiến thức và dữ liệu training cho chatbot
        \item \textit{Thuộc tính chính}:
        \begin{itemize}
            \item chatbotId (UUID): ID của chatbot
            \item name (String): Tên cơ sở kiến thức
            \item description (String): Mô tả nội dung
            \item type (KnowledgeBaseType): Loại kiến thức
            \item status (KnowledgeBaseStatus): Trạng thái hoạt động
            \item lastTrainingDate (DateTime): Thời điểm training gần nhất
            \item vectorStore (VectorStoreConfig): Cấu hình lưu trữ vector
        \end{itemize}
        \item \textit{Quan hệ}:
        \begin{itemize}
            \item documents (Set<DocumentEntity>): Tài liệu training
        \end{itemize}
        \item \textit{Chức năng chính}:
        \begin{itemize}
            \item Quản lý và tổ chức tài liệu
            \item Kiểm soát quá trình training
            \item Đảm bảo chất lượng dữ liệu
        \end{itemize}
    \end{itemize}

    \item \textbf{DocumentEntity}
    \begin{itemize}
        \item \textit{Mục đích}: Quản lý các tài liệu và nội dung training
        \item \textit{Thuộc tính chính}:
        \begin{itemize}
            \item knowledgeBaseId (UUID): ID cơ sở kiến thức
            \item title (String): Tiêu đề tài liệu
            \item content (String): Nội dung
            \item type (DocumentType): Loại tài liệu
            \item format (FileFormat): Định dạng file
            \item fileSize (Long): Kích thước file
            \item status (ProcessingStatus): Trạng thái xử lý
            \item metadata (Map<String, Object>): Metadata
            \item processingStatus (ProcessingStatus): Trạng thái xử lý
            \item errorMessage (String): Thông báo lỗi
            \item checksum (String): Mã kiểm tra
            \item uploadedBy (UUID): Người tải lên
        \end{itemize}
        \item \textit{Quan hệ}:
        \begin{itemize}
            \item chunks (List<TextChunkEntity>): Các đoạn văn bản
        \end{itemize}
        \item \textit{Chức năng chính}:
        \begin{itemize}
            \item Xử lý và phân đoạn nội dung tài liệu
            \item Kiểm tra tính hợp lệ của tài liệu
        \end{itemize}
    \end{itemize}
\end{itemize}

\textbf{Các lớp xử lý Ngôn ngữ tự nhiên}

\begin{itemize}
    \item \textbf{TextChunkEntity}
    \begin{itemize}
        \item \textit{Mục đích}: Quản lý các đoạn văn bản đã được phân đoạn để xử lý
        \item \textit{Thuộc tính chính}:
        \begin{itemize}
            \item documentId (UUID): ID tài liệu gốc
            \item content (String): Nội dung đoạn văn bản
            \item startIndex (int): Vị trí bắt đầu trong tài liệu
            \item endIndex (int): Vị trí kết thúc trong tài liệu
            \item embedding (Embedding): Vector biểu diễn
            \item relevanceScore (float): Điểm liên quan
            \item tokenCount (int): Số lượng token
        \end{itemize}
        \item \textit{Quan hệ}:
        \begin{itemize}
            \item document (DocumentEntity): Tài liệu gốc
            \item embedding (EmbeddingEntity): Vector embedding
        \end{itemize}
        \item \textit{Chức năng chính}:
        \begin{itemize}
            \item Tạo và quản lý embedding
            \item Đánh giá độ liên quan
        \end{itemize}
    \end{itemize}

    \item \textbf{EmbeddingEntity}
    \begin{itemize}
        \item \textit{Mục đích}: Quản lý các vector biểu diễn văn bản
        \item \textit{Thuộc tính chính}:
        \begin{itemize}
            \item textChunkId (UUID): ID đoạn văn bản
            \item vector (float[]): Vector embedding
            \item dimension (int): Số chiều vector
            \item model (String): Mô hình tạo embedding
            \item similarity (float): Độ tương đồng
            \item quality (float): Chất lượng embedding
            \item lastUpdated (DateTime): Thời điểm cập nhật
            \item modelVersion (String): Phiên bản mô hình
        \end{itemize}
        \item \textit{Chức năng chính}:
        \begin{itemize}
            \item Tính toán độ tương đồng
            \item Chuẩn hóa vector
            \item Kiểm tra tính hợp lệ vector
        \end{itemize}
    \end{itemize}
\end{itemize}

\textbf{Các lớp quản lý Hội thoại}

\begin{itemize}
    \item \textbf{ConversationEntity}
    \begin{itemize}
        \item \textit{Mục đích}: Quản lý các cuộc hội thoại giữa người dùng với chatbot hoặc nhân viên chăm sóc khách hàng
        \item \textit{Thuộc tính chính}:
        \begin{itemize}
            \item chatbotId (UUID): ID của chatbot
            \item agentId (UUID): ID nhân viên
            \item status (ConversationStatus): Trạng thái hội thoại
            \item startTime (DateTime): Thời điểm bắt đầu
            \item endTime (DateTime): Thời điểm kết thúc
        \end{itemize}
        \item \textit{Quan hệ}:
        \begin{itemize}
            \item messages (List<MessageEntity>): Danh sách tin nhắn
            \item agent (AgentEntity): Nhân viên phụ trách
            \item chatbot (ChatbotEntity): Chatbot xử lý
        \end{itemize}
        \item \textit{Chức năng chính}:
        \begin{itemize}
            \item Quản lý luồng hội thoại
            \item Chuyển giao hội thoại giữa chatbot và nhân viên
        \end{itemize}
    \end{itemize}

    \item \textbf{MessageEntity}
    \begin{itemize}
        \item \textit{Mục đích}: Quản lý các tin nhắn trong hội thoại
        \item \textit{Thuộc tính chính}:
        \begin{itemize}
            \item conversationId (UUID): ID cuộc hội thoại
            \item content (String): Nội dung tin nhắn
            \item senderType (SenderType): Loại người gửi
            \item sendAt (DateTime): Thời điểm gửi
            \item status (MessageStatus): Trạng thái tin nhắn
            \item attachments (List<Attachment>): Tệp đính kèm
        \end{itemize}
        \item \textit{Chức năng chính}:
        \begin{itemize}
            \item Xử lý và lưu trữ tin nhắn
            \item Quản lý tệp đính kèm
            \item Theo dõi trạng thái gửi nhận
        \end{itemize}
    \end{itemize}
\end{itemize}

\textbf{Các lớp phân tích và thống kê}

\begin{itemize}
    \item \textbf{AnalyticsEntity}
    \begin{itemize}
        \item \textit{Mục đích}: Quản lý và theo dõi các chỉ số phân tích trong hệ thống
        \item \textit{Thuộc tính chính}:
        \begin{itemize}
            \item entityType (EntityType): Loại đối tượng được phân tích
            \item entityId (UUID): ID đối tượng
            \item metrics (Map<String, Double>): Các chỉ số đo lường
            \item period (TimePeriod): Khoảng thời gian thống kê
            \item timestamp (DateTime): Thời điểm ghi nhận
            \item status (AnalyticsStatus): Trạng thái xử lý
        \end{itemize}
        \item \textit{Các loại chỉ số chính}:
        \begin{itemize}
            \item \textbf{Chatbot Metrics}:
            \begin{itemize}
                \item totalConversations: Tổng số cuộc hội thoại
                \item averageResponseTime: Thời gian phản hồi trung bình
                \item handoffRate: Tỷ lệ chuyển giao cho agent
                \item activeUsers: Số người dùng hoạt động
                \item peakUsageTime: Thời điểm sử dụng cao điểm
            \end{itemize}
            \item \textbf{Agent Metrics}:
            \begin{itemize}
                \item handledConversations: Số cuộc hội thoại đã xử lý
                \item averageHandlingTime: Thời gian xử lý trung bình
                \item firstResponseTime: Thời gian phản hồi đầu tiên
                \item resolutionRate: Tỷ lệ giải quyết vấn đề
                \item onlineTime: Thời gian trực tuyến
            \end{itemize}
            \item \textbf{Knowledge Base Metrics}:
            \begin{itemize}
                \item totalDocuments: Tổng số tài liệu
                \item averageProcessingTime: Thời gian xử lý trung bình
                \item queryAccuracy: Độ chính xác truy vấn
                \item contentCoverage: Độ bao phủ nội dung
                \item updateFrequency: Tần suất cập nhật
                \item searchLatency: Độ trễ tìm kiếm
                \item embeddingQuality: Chất lượng embedding
            \end{itemize}
        \end{itemize}
        \item \textit{Quan hệ}:
        \begin{itemize}
            \item relatedEntities (Map<EntityType, UUID>): Các entity liên quan
            \item alerts (Set<AlertConfig>): Cấu hình cảnh báo
        \end{itemize}
        \item \textit{Chức năng chính}:
        \begin{itemize}
            \item Tự động thu thập chỉ số theo thời gian thực
            \item Ghi nhận các sự kiện quan trọng
            \item Theo dõi các chỉ số hiệu suất (KPI)
        \end{itemize}
    \end{itemize}
\end{itemize}

\begin{itemize}
    \item \textbf{IBaseRepository<T extends BaseEntity>}
    \begin{itemize}
        \item \textit{Mục đích}: Interface cơ sở định nghĩa các thao tác CRUD chung
        \item \textit{Các phương thức chính}:
            \begin{itemize}
                \item findById(UUID): Optional<T>
                    \begin{itemize}
                        \item Tìm kiếm một đối tượng theo ID
                    \end{itemize}
                \item findAll(): List<T>
                    \begin{itemize}
                        \item Lấy danh sách tất cả các đối tượng
                    \end{itemize}
                \item save(T): T
                    \begin{itemize}
                        \item Lưu một đối tượng mới vào cơ sở dữ liệu
                    \end{itemize}
                \item update(T): T
                    \begin{itemize}
                        \item Cập nhật thông tin của một đối tượng
                    \end{itemize}
                \item delete(UUID): void
                    \begin{itemize}
                        \item Xóa một đối tượng theo ID
                    \end{itemize}
                \item exists(UUID): boolean
                    \begin{itemize}
                        \item Kiểm tra sự tồn tại của một đối tượng theo ID
                    \end{itemize}
                \item count(): long
                    \begin{itemize}
                        \item Đếm tổng số đối tượng trong cơ sở dữ liệu
                    \end{itemize}
            \end{itemize}
    \end{itemize}

    \item \textbf{IBaseUserRepository<T extends BaseUserEntity>}
    \begin{itemize}
        \item \textit{Mục đích}: Repository cơ sở cho các entity người dùng
        \item \textit{Các phương thức chính}:
            \begin{itemize}
                \item findByEmail(String): Optional<T>
                    \begin{itemize}
                        \item Tìm kiếm người dùng theo địa chỉ email
                    \end{itemize}
                \item findByEmailAndCompanyId(String, UUID): Optional<T>
                    \begin{itemize}
                        \item Tìm kiếm người dùng theo email và ID công ty
                    \end{itemize}
                \item updatePassword(UUID, String): void
                    \begin{itemize}
                        \item Cập nhật mật khẩu cho người dùng
                    \end{itemize}
                \item updateLastLogin(UUID): void
                    \begin{itemize}
                        \item Cập nhật thời gian đăng nhập cuối cùng
                    \end{itemize}
                \item findByLastLoginBefore(DateTime): List<T>
                    \begin{itemize}
                        \item Tìm kiếm danh sách người dùng có thời gian đăng nhập cuối trước một thời điểm
                    \end{itemize}
            \end{itemize}
    \end{itemize}

    \item \textbf{IAgentRepository extends IBaseUserRepository<AgentEntity>}
    \begin{itemize}
        \item \textit{Mục đích}: Quản lý truy xuất dữ liệu nhân viên chăm sóc khách hàng
        \item \textit{Các phương thức chuyên biệt}:
            \begin{itemize}
                \item findByCompanyId(UUID): List<AgentEntity>
                    \begin{itemize}
                        \item Lấy danh sách nhân viên thuộc một công ty
                    \end{itemize}
                \item findByStatus(AgentStatus): List<AgentEntity>
                    \begin{itemize}
                        \item Lấy danh sách nhân viên theo trạng thái hoạt động
                    \end{itemize}
                \item findAvailableAgents(UUID): List<AgentEntity>
                    \begin{itemize}
                        \item Tìm kiếm các nhân viên đang rảnh và có thể tiếp nhận công việc
                    \end{itemize}
                \item updateStatus(UUID, AgentStatus): void
