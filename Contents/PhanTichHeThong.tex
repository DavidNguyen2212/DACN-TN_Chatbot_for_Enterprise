\section{Phân tích hệ thống}
\subsection{Stakeholders}
\subsubsection{Đội phát triển phần mềm}

Nhóm phát triển phần mềm chịu trách nhiệm xây dựng, thử nghiệm và triển khai sản phẩm theo các yêu cầu được cung cấp.

\begin{itemize}
    \item \textbf{Nhu cầu:}
    \begin{itemize}
        \item Nhóm cần có phạm vi dự án và các yêu cầu được xác định rõ ràng để đảm bảo lập kế hoạch hiệu quả và đồng nhất với các mục tiêu được đặt ra.
        \item Nhóm cần được tạo điều kiện tiếp cận các công cụ và công nghệ cần thiết để thực hiện nhiệm vụ của mình một cách hiệu quả và sáng tạo.
    \end{itemize}
    \item \textbf{Ảnh hưởng:} Nhóm phát triển có ảnh hưởng lớn do nhóm tham gia trực tiếp vào quá trình thực hiện dự án. Công việc của nhóm tác động trực tiếp đến chất lượng và chức năng của sản phẩm cuối cùng, khiến cho sự đóng góp và sự hài lòng của nhóm phát triển quan trọng đối với sự thành công của dự án.
\end{itemize}

\subsubsection{Chủ doanh nghiệp}

Đây là những chủ doanh nghiệp sử dụng hệ thống để tạo và triển khai chatbot AI trên trang web của họ.

\begin{itemize}
    \item \textbf{Nhu cầu:}
    \begin{itemize}
        \item Người dùng cần một giao diện trực quan và tích hợp liền mạch để tăng hiệu quả và giảm thiểu thời gian triển khai.
        \item Chatbot cần có khả năng tùy chỉnh giao diện để phù hợp với nhận diện thương hiệu của doanh nghiệp.
        \item Chức năng của chatbot cần đáng tin cậy và hiệu suất cao để duy trì sự hài lòng của người dùng.
        \item Hệ thống cần đảm bảo bảo vệ dữ liệu và tuân thủ các quy định liên quan.
        \item Cần có kênh hỗ trợ nhanh chóng và tài liệu hướng dẫn chi tiết giúp người dùng  sử dụng nền tảng một cách hiệu quả.
        \item Giá cả rõ ràng và hợp lý giúp người dùng quản lý ngân sách hợp lý.
        \item Quản lý tổ chức và các thành viên của tổ chức
    \end{itemize}
    \item \textbf{Ảnh hưởng:} Các công ty khách hàng có ảnh hưởng lớn vì các yêu cầu của họ định hình mạnh mẽ các tính năng và chức năng của sản phẩm. Phản hồi và nhu cầu của họ định hình nên hướng phát triển của hệ thống.
\end{itemize}

\subsubsection{Quản trị viên doanh nghiệp}

Quản trị viên hỗ trợ chủ sở hữu doanh nghiệp trong việc quản lý chatbot của doanh nghiệp mình.

\begin{itemize}
    \item \textbf{Nhu cầu:}
    \begin{itemize}
        \item Nhu cầu tương tự chủ doanh nghiệp, không đòi hỏi việc quản lý tổ chức
    \end{itemize}
    \item \textbf{Ảnh hưởng:} Quản trị viên hệ thống có ảnh hưởng cao do vai trò quan trọng của họ gần như tương đương với chủ sở hữu doanh nghiệp
\end{itemize}

\subsubsection{Nhóm hỗ trợ khách hàng cho doanh nghiệp}

Các nhóm này xử lý các yêu cầu phức tạp của khách hàng và quản lý các tương tác được chuyển tiếp từ chatbot.

\begin{itemize}
    \item \textbf{Nhu cầu:}
    \begin{itemize}
        \item Hệ thống có khả năng chuyển đổi liền mạch giữa chatbot và các tác nhân con người để duy trì chất lượng dịch vụ.
        \item Cho phép truy cập vào lịch sử hội thoại và phân tích, từ đó hiểu các vấn đề của khách hàng và cải thiện dịch vụ.
        \item Việc nhắn tin với khách hàng cần đáp ứng phản hồi theo thời gian thực nhằm hỗ trợ kịp thời nhu cầu của khách hàng
    \end{itemize}
    \item \textbf{Ảnh hưởng:} Các nhóm hỗ trợ khách hàng có ảnh hưởng trung bình vì phản hồi và kinh nghiệm của họ ảnh hưởng đến khả năng sử dụng và hiệu quả của hệ thống.
\end{itemize}


\subsubsection{Người dùng cuối (Khách hàng của các doanh nghiệp khách hàng)}

Đây là những cá nhân tương tác với chatbot trên trang web của doanh nghiệp nhằm tìm kiếm thông tin hoặc có nhu cầu hỗ trợ.

\begin{itemize}
    \item \textbf{Nhu cầu:}
    \begin{itemize}
        \item Chatbot cần có khả năng phản hồi nhanh chóng và chính xác cho các truy vấn.
        \item Tương tác với chatbot được cá nhân hóa, cải thiện mức độ tương tác và trải nghiệm của người dùng.
        \item Hệ thống cần đảm bảo quyền riêng tư và bảo mật dữ liệu của người dùng.
        \item Có khả năng chuyển đối tượng giao tiếp từ AI qua con người để đảm bảo hỗ trợ toàn diện.
    \end{itemize}
    \item \textbf{Ảnh hưởng:} Người dùng cuối có ảnh hưởng trực tiếp thấp nhưng tác động gián tiếp đến dự án thông qua hành vi sử dụng và phản hồi của họ, thông tin này cũng sẽ định hình cho các cải tiến của hệ thống.
\end{itemize}


\subsubsection{Nhóm đào tạo và hỗ trợ}

Các nhóm này chịu trách nhiệm cung cấp dịch vụ hướng dẫn, đào tạo và hỗ trợ liên tục cho các công ty khách hàng.

\begin{itemize}
    \item \textbf{Nhu cầu:}
    \begin{itemize}
        \item Phát triển tài liệu hướng dẫn toàn diện, chi tiết giúp khách hàng hiểu và sử dụng nền tảng hiệu quả.
        \item Giải quyết hiệu quả các vấn đề và thắc mắc: Hỗ trợ nhanh chóng và hiệu quả là điều cần thiết để duy trì lòng tin của khách hàng.
        \item Thu thập phản hồi để cải tiến liên tục: Thu thập phản hồi giúp tinh chỉnh các quy trình đào tạo và hỗ trợ.
    \end{itemize}
    \item \textbf{Ảnh hưởng:} Các nhóm đào tạo và hỗ trợ có ảnh hưởng từ thấp đến trung bình, chủ yếu thông qua tác động của họ đến tỷ lệ áp dụng và sự hài lòng của khách hàng thông qua hiệu quả đào tạo.
\end{itemize}

\subsection{User Stories}

\subsubsection{Chủ doanh nghiệp}
\begin{itemize}
    \item \textbf{Là} chủ doanh nghiệp, \textbf{tôi muốn} quy trình đăng ký tài khoản mới an toàn với xác minh email và đăng nhập bảo mật, \textbf{để} bảo vệ thông tin công ty và truy cập hệ thống một cách an toàn.
    \item \textbf{Là} chủ doanh nghiệp, \textbf{tôi muốn} quản lý và cập nhật hồ sơ công ty, bao gồm tên, logo và thông tin liên hệ, \textbf{để} duy trì thông tin chính xác và phù hợp với thương hiệu.
    \item \textbf{Là} chủ doanh nghiệp, \textbf{tôi muốn} quản lý các thành viên của công ty, \textbf{để} các thành viên sẽ hỗ trợ công ty trong việc quản lý chatbot cũng như chặm sóc khách hàng.
    \item \textbf{Là} chủ doanh nghiệp, \textbf{tôi muốn} tải lên, quản lý và phân loại dữ liệu tri thức đa dạng, \textbf{để} xây dựng và duy trì cơ sở tri thức cho chatbot một cách hiệu quả.
    \item \textbf{Là} chủ doanh nghiệp, \textbf{tôi muốn} tùy chỉnh giao diện và hành vi của chatbot bao gồm giọng điệu, lời chào và màu sắc, \textbf{để} phù hợp với nhận diện thương hiệu và tạo trải nghiệm người dùng tốt hơn.
    \item \textbf{Là} chủ doanh nghiệp, \textbf{tôi muốn} đào tạo mô hình AI, \textbf{để} có thể cập nhật cập nhật tri thức mới lên chatbot và chatbot hoạt động hiệu quả khi đào tạo xong.
    \item \textbf{Là} chủ doanh nghiệp, \textbf{tôi muốn} tích hợp mã nhúng JavaScript vào trang web doanh nghiệp với hướng dẫn chi tiết và tối ưu hóa hiệu suất, \textbf{để} triển khai chatbot dễ dàng và không ảnh hưởng đến trải nghiệm người dùng.
    \item \textbf{Là} chủ doanh nghiệp, \textbf{tôi muốn} bảng điều khiển quản lý chatbot hiển thị dữ liệu thời gian thực và tạo báo cáo phân tích chi tiết, \textbf{để} theo dõi số liệu để có thể đánh giá và cải thiện chất lượng dịch vụ.
    \item \textbf{Là} chủ doanh nghiệp, \textbf{tôi muốn} quản lý thanh toán với các gói giá linh hoạt, xử lý thanh toán an toàn và theo dõi việc sử dụng, \textbf{để} quản lý ngân sách và chi phí hiệu quả.
    \item \textbf{Là} chủ doanh nghiệp, \textbf{tôi muốn} thiết lập giới hạn truy vấn của khách hàng, \textbf{để} tránh các trường hợp phá hoại không mong muốn, hao hụt tài nguyên của doanh nghiệp.
\end{itemize}

\subsection{Yêu cầu chức năng}

\subsubsection{Đăng ký và xác thực}
\begin{itemize}
    \item Hệ thống phải cung cấp chức năng đăng ký tài khoản mới cho tổ chức, bao gồm bước xác thực địa chỉ email.
    \item Hệ thống phải cho phép người dùng đăng nhập bằng email và mật khẩu đã đăng ký.
    \item Hệ thống phải cung cấp chức năng đặt lại mật khẩu, trong đó người dùng nhận được một liên kết qua email có giới hạn thời gian hiệu lực.
\end{itemize}

\subsubsection{Quản lý tài khoản tổ chức}
\begin{itemize}
    \item Hệ thống phải cho phép quản trị viên tổ chức cập nhật thông tin hồ sơ của tổ chức, bao gồm: tên tổ chức, logo, thông tin liên hệ và địa chỉ.
    \item Hệ thống phải cho phép quản trị viên tổ chức gửi lời mời tham gia tổ chức cho thành viên mới qua email.
    \item Hệ thống phải hỗ trợ phân quyền truy cập cho thành viên tổ chức dựa trên các vai trò được định nghĩa, bao gồm 3 vai trò: Chủ sở hữu, Quản trị viên, và Nhân viên chăm sóc khách hàng.
    \item Hệ thống phải cho phép quản trị viên tổ chức quản lý danh sách thành viên, bao gồm chức năng thêm và xóa thành viên.
\end{itemize}

\subsubsection{Tùy chỉnh giao diện và hành vi chatbot}
\begin{itemize}
    \item Hệ thống phải cho phép người dùng quản trị tùy chỉnh các yếu tố giao diện hiển thị của chatbot (ví dụ: bảng màu, logo, hình nền, phông chữ, kích thước) để đáp ứng nhận diện thương hiệu của tổ chức.
    \item Hệ thống phải cung cấp cho người dùng quản trị lựa chọn ít nhất 4 phong cách hội thoại được định nghĩa trước cho chatbot.
    \item Hệ thống phải cho phép người dùng quản trị tùy chỉnh các tin nhắn mặc định của chatbot, bao gồm lời chào và các phản hồi mặc định.
    \item Hệ thống phải cung cấp chức năng xem trước cho phép người dùng quản trị đánh giá các thay đổi về giao diện và cấu hình trước khi áp dụng.
    \item Hệ thống phải cho phép người dùng quản trị cấu hình giới hạn tần suất yêu cầu (rate limiting) đối với các tương tác của người dùng cuối với chatbot.
\end{itemize}
\subsubsection{Tạo mã nhúng chatbot}
\begin{itemize}
    \item Hệ thống phải cung cấp chức năng tạo mã nhúng JavaScript để tích hợp chatbot, đảm bảo kích thước tải về ban đầu của script không vượt quá 100KB.
    \item Hệ thống phải cho phép quản trị viên tổ chức cấu hình danh sách các tên miền được phép nhúng và hiển thị chatbot.
    \item Giao diện chatbot nhúng phải đảm bảo hiển thị và hoạt động phù hợp trên các kích thước viewport khác nhau, tối thiểu từ 320px (thiết bị di động) đến 2560px (màn hình desktop lớn).
\end{itemize}

\subsubsection{Quản lý cơ sở tri thức}
\begin{itemize}
    \item Hệ thống phải cho phép người dùng quản trị tải lên các tệp tài liệu với giới hạn kích thước tối đa 20MB cho mỗi tệp.
    \item Chức năng tải lên tài liệu phải hỗ trợ các định dạng tệp phổ biến, bao gồm: PDF, DOC, DOCX, và TXT.
    \item Hệ thống phải cung cấp các phương thức nhập liệu bổ sung cho cơ sở tri thức, bao gồm nhập văn bản trực tiếp qua giao diện và trích xuất nội dung từ địa chỉ URL được cung cấp.
    \item Hệ thống phải cung cấp giao diện quản lý cho phép người dùng quản trị thực hiện các thao tác trên cơ sở tri thức, bao gồm: thêm, chỉnh sửa và xóa tri thức.
    \item Giao diện quản lý cơ sở tri thức phải hỗ trợ chức năng tìm kiếm và lọc các đơn vị tri thức dựa trên các tiêu chí như: tên, loại hình, và mức độ ưu tiên.
    \item Hệ thống phải cho phép người dùng quản trị tải về các tệp tài liệu đã được tải lên hệ thống trước đó.
    \item Hệ thống phải đảm bảo rằng chỉ có một tiến trình huấn luyện được thực thi đồng thời cho mỗi tổ chức tại một thời điểm.
\end{itemize}

\subsubsection{Trò chuyện với nhân viên hỗ trợ trực tiếp}
\begin{itemize}
    \item Hệ thống phải hỗ trợ chức năng chuyển giao liền mạch cuộc trò chuyện từ chatbot sang nhân viên hỗ trợ trực tuyến khi có yêu cầu từ người dùng.
    \item Hệ thống phải cung cấp giao diện cho phép nhân viên hỗ trợ tiếp nhận, theo dõi và phản hồi các cuộc trò chuyện được phân công cho họ.
    \item Hệ thống phải cung cấp cho quản trị viên tổ chức quyền truy cập để xem  lịch sử hoặc nội dung các cuộc hội thoại, bao gồm cả các tương tác với chatbot và các phiên hỗ trợ trực tiếp bởi nhân viên.
    \item Hệ thống phải phân phối các yêu cầu hỗ trợ trực tiếp tới các nhân viên hỗ trợ một cách cân bằng.
    \item Hệ thống phải cho phép nhân viên hỗ trợ hoặc quản trị viên tổ chức kết thúc các phiên trò chuyện, ví dụ như khi vấn đề đã được giải quyết hoặc theo yêu cầu quản trị.
\end{itemize}

\subsubsection{Báo cáo và thống kê}
\begin{itemize}
    \item Hệ thống phải cung cấp một giao diện hiển thị các chỉ số hiệu suất và sử dụng chính, bao gồm: tổng số tin nhắn đã xử lý, tổng số phiên hội thoại (phân loại theo chatbot và nhân viên hỗ trợ), và tổng chi phí sử dụng dịch vụ. Giao diện này cần hỗ trợ lọc dữ liệu theo khoảng thời gian.
    \item Hệ thống phải cho phép quản trị viên tổ chức xuất dữ liệu thống kê chi tiết hoặc các báo cáo tổng hợp từ giao diện báo cáo.
\end{itemize}

\subsubsection{Quản lý Gói dịch vụ và Thanh toán}
\begin{itemize}
    \item Hệ thống phải cung cấp ít nhất 2 gói dịch vụ trả phí, mỗi gói có các tính năng và giới hạn sử dụng riêng biệt được mô tả chi tiết và rõ ràng.
    \item Ngoài các gói dịch vụ trả phí định kỳ, hệ thống cần hỗ trợ mô hình định giá theo mức sử dụng thực tế cho một số tài nguyên nhất định (ví dụ: dung lượng lưu trữ tài liệu, số lượng yêu cầu AI, số lượng tài khoản nhân viên hỗ trợ).
    \item Hệ thống phải cung cấp cho quản trị viên tổ chức khả năng theo dõi mức sử dụng tài nguyên hiện tại so với giới hạn của gói dịch vụ đang đăng ký.
    \item Hệ thống phải cung cấp tùy chọn đăng ký gói dùng thử miễn phí với thời hạn hoặc giới hạn sử dụng nhất định, đồng thời đảm bảo quy trình nâng cấp lên gói trả phí diễn ra thuận tiện và liền mạch cho người dùng.
    \item Hệ thống phải cung cấp cho quản trị viên tổ chức quyền truy cập vào lịch sử giao dịch và hóa đơn thanh toán chi tiết, bao gồm ngày tháng, số tiền, gói dịch vụ tương ứng và trạng thái thanh toán.
\end{itemize}


\subsection{Yêu cầu phi chức năng}
\subsubsection{Hiệu suất (Performance)}
\begin{itemize}
    \item \textbf{Thời gian phản hồi Chatbot:} Dưới điều kiện tải thông thường (10-20 người dùng đồng thời), 90\% phản hồi của chatbot đối với truy vấn người dùng phải được hoàn thành trong vòng 3 giây.
    \item \textbf{Thời gian xử lý tác vụ nền:} Thời gian trung bình để hệ thống trích xuất và xử lý nội dung từ một URL được cung cấp làm nguồn tri thức không được vượt quá 15 giây cho các trang web có kích thước thông thường (dưới 2MB HTML).
    \item \textbf{Khả năng chịu tải:} Hệ thống phải có khả năng xử lý đồng thời ít nhất 30 người dùng tương tác mà không làm tăng thời gian phản hồi trung bình của chatbot quá 50\% so với điều kiện tải thông thường.
    \item \textbf{Khả năng mở rộng:} Kiến trúc hệ thống phải được thiết kế sử dụng các thành phần có khả năng mở rộng theo chiều ngang để cho phép xử lý tải tăng lên trong tương lai bằng cách bổ sung tài nguyên máy chủ, thay vì yêu cầu thiết kế lại toàn bộ hệ thống.
\end{itemize}

\subsubsection{Tính khả dụng (Availability)}
\begin{itemize}
    \item \textbf{Độ sẵn sàng:} Hệ thống được thiết kế hướng tới mục tiêu độ sẵn sàng 99.5\% trong giờ hoạt động dự kiến, không bao gồm thời gian bảo trì theo lịch trình.    
    \item \textbf{Bảo trì theo lịch trình:} Mọi hoạt động bảo trì yêu cầu ngừng hệ thống phải được thông báo trước ít nhất 24 giờ và thực hiện trong khung giờ có lượng truy cập thấp. Tổng thời gian bảo trì theo lịch trình không vượt quá 4 giờ mỗi tháng.
\end{itemize}

\subsubsection{Bảo mật (Security)}
\begin{itemize}
    \item \textbf{Chính sách mật khẩu:} Hệ thống phải thực thi chính sách mật khẩu mạnh, đảm bảo mật khẩu có độ dài tối thiểu 8 ký tự, không quá giống với thuộc tính người dùng, không nằm trong danh sách mật khẩu phổ biến, và không chỉ chứa toàn số. Hệ thống phải băm mật khẩu trước khi lưu trữ.
    \item \textbf{Kiểm soát truy cập:} Hệ thống phải triển khai Kiểm soát truy cập dựa trên vai trò với ít nhất 3 vai trò được định nghĩa (ví dụ: Chủ sở hữu, Quản trị viên, Nhân viên), đảm bảo người dùng chỉ có thể truy cập các chức năng và dữ liệu phù hợp với vai trò được gán.
    \item \textbf{Mã hóa dữ liệu truyền:} Toàn bộ giao tiếp giữa trình duyệt người dùng và máy chủ hệ thống phải được mã hóa bằng TLS 1.2 hoặc cao hơn.
    \item \textbf{Mã hóa dữ liệu lưu trữ:} Tất cả thông tin định danh cá nhân của người dùng và thông tin nhạy cảm của tổ chức phải được mã hóa khi lưu trữ trong cơ sở dữ liệu sử dụng thuật toán mã hóa đối xứng mạnh.
    \item \textbf{Ghi log bảo mật:} Hệ thống phải ghi log các sự kiện bảo mật quan trọng, bao gồm: đăng nhập thành công/thất bại, thay đổi mật khẩu, thay đổi quyền, tạo/xóa tài khoản quản trị. Log phải bao gồm dấu thời gian, định danh người dùng, địa chỉ IP nguồn, và mô tả sự kiện. Log phải được lưu trữ an toàn trong tối thiểu 90 ngày.
\end{itemize}

\subsubsection{Khả năng sử dụng (Usability)}
\begin{itemize}
    \item \textbf{Hiệu quả tác vụ:} Mục tiêu là ít nhất 80\% người dùng quản trị lần đầu, sau khi xem tài liệu hoặc video hướng dẫn dưới 15 phút, có thể hoàn thành thành công các tác vụ cốt lõi sau đây trong vòng 10 phút cho mỗi tác vụ: Tải lên thành công một tệp tài liệu và khởi tạo quá trình xử lý/huấn luyện; thay đổi thành công các thuộc tính giao diện chatbot và lưu lại cấu hình; và hoàn tất việc gửi lời mời tham gia tổ chức đến một địa chỉ email mới.
    \item \textbf{Tính nhất quán giao diện:} Các yếu tố điều hướng chính, thuật ngữ, và giao diện phải nhất quán trên tất cả các màn hình quản trị của hệ thống.
    \item \textbf{Khả năng truy cập:} Giao diện chatbot nhúng và các trang quản trị chính phải tuân thủ Web Content Accessibility Guidelines (WCAG) 2.1 Mức AA cho các tiêu chí sau: độ tương phản màu (4.5:1 cho văn bản thường, 3:1 cho văn bản lớn).
\end{itemize}

\subsubsection{Độ tin cậy (Reliability)}
\begin{itemize}
    \item \textbf{Kiểm tra tính hợp lệ đầu vào:} Hệ thống phải thực hiện kiểm tra tính hợp lệ cho tất cả các dữ liệu đầu vào từ người dùng trước khi xử lý. Phải ngăn chặn dữ liệu không hợp lệ hoặc không đúng định dạng gây ra lỗi hoặc làm hỏng dữ liệu. Các lỗi xác thực phải được trả về rõ ràng cho người dùng.
    \item \textbf{Xử lý lỗi và ghi log:} 100\% các lỗi không mong muốn ở phía máy chủ phải được ghi log với đầy đủ thông tin stack trace và ngữ cảnh để phục vụ việc sửa chữa. Người dùng cuối không được nhìn thấy thông tin lỗi chi tiết hoặc stack trace, thay vào đó là một thông báo lỗi thân thiện và mã tham chiếu.
\end{itemize}

\subsubsection{Khả năng bảo trì (Maintainability)}
\begin{itemize}
    \item \textbf{Chất lượng mã nguồn:} Mã nguồn phía máy chủ phải tuân thủ theo hướng dẫn về phong cách lập trình của ngôn ngữ/framework được sử dụng. Phải sử dụng công cụ phân tích mã tĩnh tích hợp trong quy trình phát triển để phát hiện các vấn đề về phong cách và lỗi tiềm ẩn.
    \item \textbf{Tính mô-đun:} Hệ thống phải được cấu trúc thành các mô-đun logic rõ ràng (ví dụ: quản lý tài khoản, quản lý tri thức, xử lý hội thoại, giao diện nhúng) với các giao diện được định nghĩa tốt giữa chúng để giảm thiểu sự phụ thuộc chéo.
    \item \textbf{Độ bao phủ kiểm thử:} Các unit test tự động phải đạt được độ bao phủ mã nguồn tối thiểu 70\% cho các mô-đun logic nghiệp vụ cốt lõi.
\end{itemize}

\subsubsection{Tính di động (Portability)}
\begin{itemize}
    \item \textbf{Tương thích trình duyệt:} Giao diện quản trị và chatbot nhúng phải hiển thị và hoạt động chính xác trên các phiên bản ổn định mới nhất của Google Chrome, Mozilla Firefox, Apple Safari, và Microsoft Edge tại thời điểm kiểm thử cuối cùng.
    \item \textbf{Thiết kế responsive:} Giao diện quản trị và chatbot nhúng phải tự động điều chỉnh bố cục để có thể sử dụng được trên các kích thước màn hình từ 360px chiều rộng đến 1920px chiều rộng.
\end{itemize}

\subsubsection{Yêu cầu Pháp lý và Đạo đức (Legal \& Ethical Requirements)}
\begin{itemize}
    \item \textbf{Tuân thủ Quyền riêng tư:} Hệ thống phải tuân thủ các yêu cầu của Nghị định 13/2023/NĐ-CP về bảo vệ dữ liệu cá nhân của Việt Nam. Điều này bao gồm việc cung cấp chính sách bảo mật rõ ràng, cơ chế lấy sự đồng ý của người dùng trước khi thu thập dữ liệu cá nhân, và quy trình xử lý yêu cầu của chủ thể dữ liệu trong vòng 30 ngày.
    \item \textbf{Minh bạch AI:} Giao diện chatbot phải hiển thị rõ ràng cho người dùng rằng họ đang tương tác với một hệ thống trí tuệ nhân tạo chứ không phải con người, ví dụ thông qua một thông báo ban đầu hoặc nhãn hiệu cố định trong giao diện trò chuyện.
\end{itemize}
